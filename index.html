<body>

  <h1>AEVRA</h1>

  <p>Tap both rings to unlock your vault</p>


  <button onclick="startScan()">Start Scanning</button>


  <div class="status">

    <div id="ringA" class="key">Ring A: waiting</div>

    <div id="ringB" class="key">Ring B: waiting</div>

    <p id="message"></p>

  </div>


  <script>

    let partA = null;

    let partB = null;


    async function startScan() {

      if (!('NDEFReader' in window)) {

        document.getElementById('message').innerText = 

          'NFC not supported. Use Chrome on Android.';

        return;

      }

      try {

        const reader = new NDEFReader();

        await reader.scan();

        document.getElementById('message').innerText = 'Scanning... tap a ring.';


        reader.onreading = (event) => {

          for (const record of event.message.records) {

            const text = new TextDecoder().decode(record.data);

            

            if (text.startsWith('aevra-partA:')) {

              partA = text.replace('aevra-partA:', '');

              document.getElementById('ringA').innerText = 'Ring A: found ✓';

              document.getElementById('ringA').classList.add('found');

            }

            if (text.startsWith('aevra-partB:')) {

              partB = text.replace('aevra-partB:', '');

              document.getElementById('ringB').innerText = 'Ring B: found ✓';

              document.getElementById('ringB').classList.add('found');

            }


            if (partA && partB) {

              const fullUrl = `https://aevra.app/vault?key=${partA}${partB}`;

              document.getElementById('message').innerText = 

                'Both rings detected! Opening vault...';

              setTimeout(() => { window.location.href = fullUrl; }, 1500);

            }

          }

        };

      } catch (err) {

        document.getElementById('message').innerText = 'Error: ' + err.message;

      }

    }

  </script>
